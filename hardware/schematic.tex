% MCU pin helpers: \lpin{y}{name}  /  \rpin{y}{name}
\newcommand{\lpin}[2]{\draw[thick] (-0.6,#1) -- (0,#1);
  \node[pin, anchor=west] at (0.1,#1) {#2};}
\newcommand{\rpin}[2]{\draw[thick] (4.5,#1) -- (5.1,#1);
  \node[pin, anchor=east] at (4.4,#1) {#2};}

% Relay module: \relay{y of IN}{index}{GPIO label}
\newcommand{\relay}[3]{
  \draw[mod] (10,#1-0.6) rectangle (14.4,#1+0.6);
  \node[title] at (12.2,#1+0.2) {K#2 \; RELAY #2 (5 V)};
  \node[pin]   at (12.2,#1-0.28) {COM / NO / NC $\to$ Load #2};
  % IN (left)
  \draw[s] (5.1,#1) -- (10,#1);
  \node[pin, sig, anchor=south] at (8.6,#1) {#3 $\to$ IN};
  % VCC (right, to +5V rail)
  \draw[p5] (14.4,#1+0.3) -- (18,#1+0.3) node[circ]{};
  \node[pin, anchor=south west] at (14.45,#1+0.3) {VCC};
  % GND (right, to common GND)
  \draw[pg] (14.4,#1-0.3) -- (16,#1-0.3) node[ground]{};
  \node[pin, anchor=north west] at (14.45,#1-0.3) {GND};
}
